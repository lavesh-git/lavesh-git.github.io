# 5.5 String Buffer

Use StringBuffer for thread-safe operations.

Use StringBuilder for high-performance string manipulations when thread safety is not a concern.

// ===============================

// STRINGBUFFER INTERNAL NOTES

// ===============================

// 🔹 StringBuffer class (actual class is thin wrapper)

public final class StringBuffer extends AbstractStringBuilder

implements java.io.Serializable, CharSequence {

// Default constructor → capacity = 16

public StringBuffer() {

super(16);

}

// If initialized with string → capacity = length + 16

public StringBuffer(String str) {

super(str.length() + 16);

append(str);

}

// 🔥 append is synchronized → thread-safe

public synchronized StringBuffer append(String str) {

toStringCache = null;   // invalidate cached string

super.append(str);      // call parent logic

return this;

}

// 🔥 insert is also synchronized

public synchronized StringBuffer insert(int offset, String str) {

super.insert(offset, str);

return this;

}

// 🔥 toString creates NEW String (immutability boundary)

public synchronized String toString() {

return new String(value, 0, count);

}

}
